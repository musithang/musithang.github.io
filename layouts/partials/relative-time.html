{{- /* Calendar-day difference (UTC day boundaries), not raw elapsed hours - a post from yesterday afternoon must still read "yesterday" even if less than 24h have actually elapsed by the time this builds. */ -}} {{- $day := time.ParseDuration "24h" -}} {{- $postMidnight := .Truncate $day -}} {{- $nowMidnight := now.Truncate $day -}} {{- $days := div (sub $nowMidnight.Unix $postMidnight.Unix) 86400 -}} {{- if le $days 0 -}}today{{- else if eq $days 1 -}}yesterday{{- else -}}{{ $days }} days ago{{- end -}}
